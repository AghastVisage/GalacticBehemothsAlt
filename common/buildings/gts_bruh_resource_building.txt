building_gts_bruh_micro_1 = {
	base_buildtime = @b1_time
	category = resource

	prerequisites = { tech_gts_micro_1 }

	potential = {
		exists = owner
		owner = {
			gts_is_giantess_empire = yes
		}
	}

	destroy_trigger = {
		exists = owner
		owner = {
			gts_is_giantess_empire = no
		}
	}

	building_sets = {
		gts_power
	}

	planet_modifier = {
		job_gts_bruh_micro_worker_1_add = 1200
		pop_gts_bruh_micro_worker_1_workforce_mult = 0.2
		pop_gts_drone_gts_bruh_giantess_fluid_worker_1_workforce_mult = 0.2
	}

	resources = {
		category = planet_buildings
		cost = {
			minerals = @b2_minerals
		}
		upkeep = {
			energy = @b2_upkeep
		}
	}

	on_built = { planet = { planet_event = { id = gts_evt_random.010300 } } }
}
