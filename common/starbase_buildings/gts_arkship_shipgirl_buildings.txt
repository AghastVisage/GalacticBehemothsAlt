gts_arkship_gateway_0 = {
	icon = "GFX_gts_gateway_0_build_icon"
	starbase_type = arkship
	construction_days = 720
	category = sm_utility_cat
	starbase_limit = { base = 1 }
	empire_limit = { base = 1 }
	potential = {
		is_arkship_starbase = yes
		owner = {
			is_nomadic = yes
			has_country_flag = gts_gateway_can_build
		}
	}
	possible = {
		custom_tooltip = {
			fail_text = requires_no_existing_gts_gateway
			owner = {
				NOR = {
					has_megastructure = gts_gateway_0
					has_megastructure = gts_gateway_final
					any_owned_ship = {
						exists = starbase
						starbase = {
							OR = {
								has_starbase_building = gts_arkship_gateway_final
								is_starbase_building_building = gts_arkship_gateway_final
							}
						}
					}
				}
			}
		}
	}
	resources = {
		category = starbase_arkship_upgrades
		cost = { energy = 10000 alloys = 5000 }
		upkeep = { energy = 20 }
	}
	on_finished = {
		owner = { country_event = { id = gts_leader_interact.104 } }
	}
	custom_tooltip = sm_gts_arkship_gateway_0_tt
	ai_weight = { weight = 0 }
}
