tr_aptitude_finish = {

	on_enabled = {
		hidden_effect = {
			every_owned_leader = {
				limit = {
					is_leader_tier = leader_tier_default
				}
				add_trait = {
					trait = random_common
					show_message = yes
				}
			}
			every_pool_leader = {
				add_trait = {
					trait = random_common
					show_message = no
				}
			}
		}
	}

	modifier = {
		leader_initial_traits_add = 1
		ascension_perks_add = 1
	}
}
