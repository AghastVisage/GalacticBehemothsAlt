gts_country_is_bigger_than = {
	is_scope_type = country
	exists = $TARGET$

	calc_true_if = {
		amount = 1
		has_policy_flag = gts_empire_type_nano
		has_policy_flag = gts_empire_type_micro
		has_policy_flag = gts_empire_type_tiny
		has_policy_flag = gts_empire_type_small
		has_policy_flag = gts_empire_type_normal
		has_policy_flag = gts_empire_type_tall
		has_policy_flag = gts_empire_type_giant
		has_policy_flag = gts_empire_type_titan
		has_policy_flag = gts_empire_type_ultra
	}

	$TARGET$ = {
		is_scope_type = country
		calc_true_if = {
			amount = 1
			has_policy_flag = gts_empire_type_nano
			has_policy_flag = gts_empire_type_micro
			has_policy_flag = gts_empire_type_tiny
			has_policy_flag = gts_empire_type_small
			has_policy_flag = gts_empire_type_normal
			has_policy_flag = gts_empire_type_tall
			has_policy_flag = gts_empire_type_giant
			has_policy_flag = gts_empire_type_titan
			has_policy_flag = gts_empire_type_ultra
		}
	}

	switch = {
		trigger = has_policy_flag
		gts_empire_type_ultra = {
			$TARGET$ = {
				OR = {
					has_policy_flag = gts_empire_type_titan
					has_policy_flag = gts_empire_type_giant
					has_policy_flag = gts_empire_type_tall
					has_policy_flag = gts_empire_type_normal
					has_policy_flag = gts_empire_type_small
					has_policy_flag = gts_empire_type_tiny
					has_policy_flag = gts_empire_type_micro
					has_policy_flag = gts_empire_type_nano
				}
			}
		}
		gts_empire_type_titan = {
			$TARGET$ = {
				OR = {
					has_policy_flag = gts_empire_type_giant
					has_policy_flag = gts_empire_type_tall
					has_policy_flag = gts_empire_type_normal
					has_policy_flag = gts_empire_type_small
					has_policy_flag = gts_empire_type_tiny
					has_policy_flag = gts_empire_type_micro
					has_policy_flag = gts_empire_type_nano
				}
			}
		}
		gts_empire_type_giant = {
			$TARGET$ = {
				OR = {
					has_policy_flag = gts_empire_type_tall
					has_policy_flag = gts_empire_type_normal
					has_policy_flag = gts_empire_type_small
					has_policy_flag = gts_empire_type_tiny
					has_policy_flag = gts_empire_type_micro
					has_policy_flag = gts_empire_type_nano
				}
			}
		}
		gts_empire_type_tall = {
			$TARGET$ = {
				OR = {
					has_policy_flag = gts_empire_type_normal
					has_policy_flag = gts_empire_type_small
					has_policy_flag = gts_empire_type_tiny
					has_policy_flag = gts_empire_type_micro
					has_policy_flag = gts_empire_type_nano
				}
			}
		}
		gts_empire_type_normal = {
			$TARGET$ = {
				OR = {
					has_policy_flag = gts_empire_type_small
					has_policy_flag = gts_empire_type_tiny
					has_policy_flag = gts_empire_type_micro
					has_policy_flag = gts_empire_type_nano
				}
			}
		}
		gts_empire_type_small = {
			$TARGET$ = {
				OR = {
					has_policy_flag = gts_empire_type_tiny
					has_policy_flag = gts_empire_type_micro
					has_policy_flag = gts_empire_type_nano
				}
			}
		}
		gts_empire_type_tiny = {
			$TARGET$ = {
				OR = {
					has_policy_flag = gts_empire_type_micro
					has_policy_flag = gts_empire_type_nano
				}
			}
		}
		gts_empire_type_micro = {
			$TARGET$ = {
				has_policy_flag = gts_empire_type_nano
			}
		}
		gts_empire_type_nano = {
			always = no
		}
	}
}
