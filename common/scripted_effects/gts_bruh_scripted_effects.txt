#胖次星球改造
gts_bruh_panties_planet_change = {
	set_planet_flag = gts_reformed_planet
	change_pc = pc_gts_bruh_panties_planet_01
	set_colony_type = col_gts_bruh_panties
	validate_planet_buildings_and_districts = yes

	hidden_effect = {
		export_trigger_value_to_variable = {
			trigger = planet_size
			variable = gts_bruh_get_planet_size
		}
		set_variable_to_random_value = {
			which = gts_bruh_random_count
			min = 9
			max = gts_bruh_get_planet_size
		}
		add_deposit = d_gts_bruh_panties_planet_06_deposits
		random_list = {
			1 = { add_deposit = d_gts_bruh_panties_planet_04_deposits }
			1 = { }
		}
		while = {
			count = gts_bruh_random_count
			random_list = {
				1 = {
					add_deposit = d_gts_bruh_panties_planet_01_deposits
				}
				1 = {
					add_deposit = d_gts_bruh_panties_planet_02_deposits
				}
				1 = {
					add_deposit = d_gts_bruh_panties_planet_03_deposits
				}
			}
			random_list = {
				3 = { }
				1 = {
					add_deposit = d_gts_bruh_panties_planet_05_deposits
				}
				1 = {
					add_deposit = d_gts_bruh_panties_planet_07_deposits
