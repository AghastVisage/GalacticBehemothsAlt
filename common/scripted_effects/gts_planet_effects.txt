# 创建一个星球障碍的随机数
create_gts_planet_random_variable = {
	export_trigger_value_to_variable = {
		trigger = planet_size
		variable = gts_bruh_get_planet_size
	}
	set_variable_to_random_value = {
		which = gts_bruh_random_count
		min = 9
		max = gts_bruh_get_planet_size
	}
}
