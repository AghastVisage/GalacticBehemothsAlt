rainna_la_grande_update_debt_slot = {
    if = {
        limit = {
            has_country_flag = rainna_la_grande_valid_debt_slot_$SLOT$
        }

        # 当前总欠款进行一次月复利
        multiply_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            value = rainna_la_grande_debt_slot_$SLOT$_actual_interest_value
        }

        # 当前总欠款保留两位小数
        multiply_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            value = 100
        }

        round_variable = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value

        divide_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            value = 100
        }

        # 当前累计利息 = 当前总欠款 - 当前剩余本金
        set_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
        }

        subtract_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
        }

        # 累计利息也保留两位小数
        multiply_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = 100
        }

        round_variable = rainna_la_grande_debt_slot_$SLOT$_interest_created_value

        divide_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = 100
        }

        # 防止浮点误差导致负利息
        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                    value < 0
                }
            }

            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                value = 0
            }
        }

        # 最低贷款时间逐月减少
        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_lock_months
                    value > 0
                }
            }

            subtract_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_lock_months
                value = 1
            }
        }

        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_due_months
                    value < 0
                }
            }
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_due_months
                value = 0
            }
        }

        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_due_months
                    value > 0
                }
            }

            subtract_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_due_months
                value = 1
            }
        }
        else_if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_due_months
                    value <= 0
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = rainna_la_grande_debt_slot_$SLOT$_due_months
                        value < 0
                    }
                }
                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_due_months
                    value = 0
                }
            }
            change_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_overdue_months
                value = 1
            }
        }
    }
}

    remove_country_flag = rainna_la_grande_slot_2_selected
    remove_country_flag = rainna_la_grande_slot_3_selected
    remove_country_flag = rainna_la_grande_slot_4_selected
}

rainna_la_grande_select_debt_slot = {
    # 防止同时选中两个槽位
    rainna_la_grande_clear_debt_selection = yes

    set_country_flag = rainna_la_grande_slot_$SLOT$_selected

    # 当前所选贷款的总欠款
    set_variable = {
        which = rainna_la_grande_debt_this_slot_principal_payment_value
        value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
    }

    # 当前所选贷款的剩余本金
    set_variable = {
        which = rainna_la_grande_debt_this_slot_remaining_principal_value
        value = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
    }

    # 当前所选贷款的累计利息
    set_variable = {
        which = rainna_la_grande_debt_this_slot_interest_created_value
        value = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
    }

    # 贷款剩余锁定月份
    set_variable = {
        which = rainna_la_grande_debt_this_slot_lock_months
        value = rainna_la_grande_debt_slot_$SLOT$_lock_months
    }

    set_variable = {
        which = rainna_la_grande_debt_this_slot_due_months
        value = rainna_la_grande_debt_slot_$SLOT$_due_months
    }

    set_variable = {
        which = rainna_la_grande_debt_this_slot_overdue_months
        value = rainna_la_grande_debt_slot_$SLOT$_overdue_months
    }

    # 槽位中的利率是复利乘数，例如1.0138
    # 转换成显示百分比1.38
    set_variable = {
        which = rainna_la_grande_debt_this_slot_interest_value
        value = rainna_la_grande_debt_slot_$SLOT$_actual_interest_value
    }

    subtract_variable = {
        which = rainna_la_grande_debt_this_slot_interest_value
        value = 1
    }

    multiply_variable = {
        which = rainna_la_grande_debt_this_slot_interest_value
        value = 10000
    }

    round_variable = rainna_la_grande_debt_this_slot_interest_value

    divide_variable = {
        which = rainna_la_grande_debt_this_slot_interest_value
        value = 100
    }
}

rainna_la_grande_repay_debt_slot = {
    if = {
        limit = {
            has_country_flag = rainna_la_grande_valid_debt_slot_$SLOT$

            check_variable = {
                which = rainna_la_grande_debt_repayment_value
                value > 0
            }
        }

        # 获取玩家现有的蕾恩娜币
        export_resource_stockpile_to_variable = {
            resource = rainna_la_grande_coin
            variable = rainna_la_grande_coin_stockpile_value
        }

        # 实际准备偿还的金额
        set_variable = {
            which = rainna_la_grande_temp_repayment_value
            value = rainna_la_grande_debt_repayment_value
        }

        # 不能偿还超过该笔贷款的总欠款
        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_temp_repayment_value
                    value > rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                }
            }

            set_variable = {
                which = rainna_la_grande_temp_repayment_value
                value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            }
        }

        # 不能偿还超过玩家实际持有的蕾恩娜币
        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_temp_repayment_value
                    value > rainna_la_grande_coin_stockpile_value
                }
            }

            set_variable = {
                which = rainna_la_grande_temp_repayment_value
                value = rainna_la_grande_coin_stockpile_value
            }
        }

        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_temp_repayment_value
                    value > 0
                }
            }

            # 扣除蕾恩娜币
            add_resource = {
                rainna_la_grande_coin = -1
                multiplier = rainna_la_grande_temp_repayment_value
            }

            # 减少该笔贷款总欠款
            subtract_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                value = rainna_la_grande_temp_repayment_value
            }

            # 当前总欠款保留两位小数
            multiply_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                value = 100
            }

            round_variable = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value

            divide_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                value = 100
            }

            # 如果总欠款已经低于剩余本金，
            # 说明利息已经还完，开始偿还本金
            if = {
                limit = {
                    check_variable = {
                        which = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
                        value > rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                    }
                }

                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
                    value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                }
            }

            # 重新计算该笔贷款的累计利息
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            }

            subtract_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                value = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
            }

            multiply_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                value = 100
            }

            round_variable = rainna_la_grande_debt_slot_$SLOT$_interest_created_value

            divide_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                value = 100
            }

            if = {
                limit = {
                    check_variable = {
                        which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                        value < 0
                    }
                }

                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                    value = 0
                }
            }

            # 总欠款为0时清空贷款槽位
            if = {
                limit = {
                    check_variable = {
                        which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                        value <= 0
                    }
                }

                remove_country_flag = rainna_la_grande_valid_debt_slot_$SLOT$

                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                    value = 0
                }

                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
                    value = 0
                }

                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                    value = 0
                }

                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_actual_interest_value
                    value = 0
                }

                set_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_lock_months
                    value = 0
                }
            }

            rainna_la_grande_rebuild_debt_totals = yes
        }
    }
}

# 按百分比直接减免一笔贷款
# SLOT = 1~4
# MIN / MAX 使用百分数，例如 MIN = 10 MAX = 25 表示减免10%~25%
rainna_la_grande_reduce_debt_slot = {
    if = {
        limit = {
            has_country_flag = rainna_la_grande_valid_debt_slot_$SLOT$
            check_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                value > 0
            }
            check_variable = {
                which = rainna_la_grande_debt_slot_3_overdue_months
                value > 1
            }
        }
        # 随机生成减免百分比，例如10~25
        set_variable_to_random_value = {
            which = rainna_la_grande_temp_debt_reduction_percent
            min = $MIN$
            max = $MAX$
            rounded = yes
        }
        # 百分数转换为小数
        # 例如 25 → 0.25
        divide_variable = {
            which = rainna_la_grande_temp_debt_reduction_percent
            value = 100
        }
        # 实际减免额 = 当前总欠款 × 减免比例
        set_variable = {
            which = rainna_la_grande_temp_debt_reduction_value
            value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
        }
        multiply_variable = {
            which = rainna_la_grande_temp_debt_reduction_value
            value = rainna_la_grande_temp_debt_reduction_percent
        }
        # 实际减免额保留两位小数
        multiply_variable = {
            which = rainna_la_grande_temp_debt_reduction_value
            value = 100
        }
        round_variable = rainna_la_grande_temp_debt_reduction_value
        divide_variable = {
            which = rainna_la_grande_temp_debt_reduction_value
            value = 100
        }
        # 直接从当前总欠款中减掉
        # 注意：这里不扣玩家任何资源
        subtract_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            value = rainna_la_grande_temp_debt_reduction_value
        }
        # 当前总欠款统一保留两位小数
        multiply_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            value = 100
        }
        round_variable = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
        divide_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            value = 100
        }
        # 和正常还款一样：
        # 如果总欠款已经低于剩余本金，
        # 说明利息已经全部被抵掉，开始抵本金
        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
                    value > rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                }
            }
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
                value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
            }
        }
        # 重新计算未偿利息
        # 利息 = 当前总欠款 - 剩余本金
        set_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
        }
        subtract_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
        }
        # 利息保留两位小数
        multiply_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = 100
        }
        round_variable = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
        divide_variable = {
            which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
            value = 100
        }
        # 防止浮点误差出现负利息
        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                    value < 0
                }
            }
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                value = 0
            }
        }
        # 如果被100%减免，直接清空贷款
        if = {
            limit = {
                check_variable = {
                    which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                    value <= 0
                }
            }
            remove_country_flag = rainna_la_grande_valid_debt_slot_$SLOT$
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_principal_payment_value
                value = 0
            }
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_remaining_principal_value
                value = 0
            }
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_interest_created_value
                value = 0
            }
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_actual_interest_value
                value = 0
            }
            set_variable = {
                which = rainna_la_grande_debt_slot_$SLOT$_lock_months
                value = 0
            }
        }
        # 重新统计银行总账
        rainna_la_grande_rebuild_debt_totals = yes
    }
}
