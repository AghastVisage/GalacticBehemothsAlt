#         ETHIC = materialist                           # 思潮
#         EVENT_LEADER = yes                            # 是否为事件领袖 仅影响能否成为统治者

#         DESC = DESC_gts_XXXX                          # 领袖的描述
#         CATCH_PHRASE = gts_XXXX_catch_phrase          # 获取领袖时的播报讯息

#         RANDOMIZE_TRAITS = no                         # 是否使用随机特质
#         TARGET = gts_XXXX                             # 领袖目标 请固定使用gts_XXX的格式
#         TRAIT1 = gts_trait_XXXX1                      # 设置特殊特质，目前最高上限仅支持10个
#         TRAIT2 = gts_trait_XXXX2                      #
#         TRAIT3 = gts_trait_XXXX4                      #
#         TRAIT4 = gts_trait_XXXX6                      #

# 犽戎招募-2
gts_leader_yarong_hired2 = {
    create_gts_friendly_country = {
        name = gts_yarong
        icon_category = spherical
        icon_file = flag_spherical_6.dds
    }
    if = {
        limit = {
            exists = event_target:gts_yarong
        }
        event_target:gts_yarong = { set_owner = root }
    }
    else = {
        gts_leader_yarong_hired = yes
        gts_yarong_add_trait = yes
        gts_leader_hire_effect = yes
    }

    call_gts_fleet = {
        class = NAME_gts_yarong_ship
        name = gts_yarong
    }
    gts_yarong_change_design = yes
    event_target:gts_yarong = {
        gts_love_value_set = { base_value = 50 increase_value = 10 }
    }
    save_event_target_as = gts_yarong_owned_country
}
