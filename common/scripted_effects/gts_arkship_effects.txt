gts_create_seele_arkship_effect = {
	if = {
		limit = {
			NOT = {
				any_owned_ship = {
					OR = {
						is_ship_size = gts_seele_1_swallowtail_phantasm
						is_ship_size = gts_seele_2_stygian_nymph
						is_ship_size = gts_seele_3_herrscher_of_rebirth
						is_ship_size = gts_seele_4_wings_of_panacea
					}
				}
			}
		}
		# set_ship_design = { design = NAME_gts_bronya_3_silverwing_n_ex }
		create_fleet = {
			name = gts_seele
			effect = {
				set_owner = root
				create_ship = {
					name = gts_seele
					design = NAME_gts_seele_1_swallowtail_phantasm
					graphical_culture = civilian_arkship
					effect = {
						add_modifier = {
							modifier = natural_beauty
						}
						add_modifier = {
							modifier = atmospheric_aphrodisiac
						}
						add_modifier = {
							modifier = atmospheric_hallucinogen
						}
						add_modifier = {
							modifier = ultra_rich
						}
						add_modifier = {
							modifier = hazardous_weather
						}
						add_district = district_bfbf_seele_0
						save_global_event_target_as = gts_seele_arkship
						set_ship_flag = gts_body_ship
						initialize_arkship_starbase_effect = yes
					}
				}
				set_location = {
					target = root.capital_scope
					distance = 10
					angle = 90
					direction = in_system
				}
			}
		}
	}
}
