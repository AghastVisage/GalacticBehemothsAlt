@love_level2 = 25
@love_level3 = 50
@love_level4 = 75
@love_level5 = 100
@gts_normal_love_reduce = 3
@gts_normal_love_increase_1 = 4
@gts_normal_love_increase_2 = 5
@gts_normal_love_increase_3 = 6

# 借鉴至舰Rmod好感度系统
# 常规好感度系统启动，如果需要设置特殊好感度系统，请在领袖招募时增加set_leader_flag = not_has_gts_love_trait_normal
gts_love_normal_system_launch = {
    optimize_memory
    if = {
        limit = {
            NOR = {
                has_leader_flag = has_gts_love_trait
                has_leader_flag = not_has_gts_love_trait_normal
            }
        }
        set_leader_flag = has_gts_love_trait
        set_leader_flag = has_gts_love_trait_normal
        set_variable = { which = gts_love_logs value = 0 }
        set_variable = { which = gts_love_level2 value = @love_level2 }
        set_variable = { which = gts_love_level3 value = @love_level3 }
        set_variable = { which = gts_love_level4 value = @love_level4 }
        set_variable = { which = gts_love_level5 value = @love_level5 }
        set_variable = { which = gts_normal_love_reduce  value = @gts_normal_love_reduce }
        random_list = {
            60 = { set_variable = { which = gts_normal_love_increase value = @gts_normal_love_increase_1 } }
            20 = { set_variable = { which = gts_normal_love_increase value = @gts_normal_love_increase_2 } }
            20 = { set_variable = { which = gts_normal_love_increase value = @gts_normal_love_increase_3 } }
        }
        if = { limit = { NOT = { has_trait = trait_leader_gts_love_logs_normal } }
            add_trait = { trait = trait_leader_gts_love_logs_normal show_message = no }
        }
    }
}
# 常规好感度检测
gts_love_normal_system_check = {
    optimize_memory
    if = {
        limit = {
            is_gts_love_leader = yes
        }
        if = {
            limit = {
                NOT = {
                    has_trait = trait_leader_gts_love_logs_normal
                }
            }
            add_trait = { trait = trait_leader_gts_love_logs_normal show_message = no }
        }
    }
    if = {
        limit = { has_leader_flag = not_has_gts_love_trait_normal }
        gts_love_normal_system_end = yes
    }
}
#好感度增长 可以增加特定值
gts_love_system_increase = {
    optimize_memory
    if = {
        limit = {
            is_gts_love_leader = yes
        }
        if = {
            limit = {
                check_variable = { which = gts_love_logs value < gts_love_level5 }
            }
            change_variable = { which = gts_love_logs value = $value|gts_normal_love_increase$ }
        }
        if = {
            limit = {
                check_variable = { which = gts_love_logs value > gts_love_level5 }
            }
            set_variable = { which = gts_love_logs value = gts_love_level5 }
        }
    }
}
#好感度减少 可以减去特定值
gts_love_system_reduce = {
    optimize_memory
    if = {
        limit = {
            is_gts_love_leader = yes
        }
        subtract_variable = { which = gts_love_logs value = $value|gts_normal_love_reduce$ }
        if = {
            limit = { check_variable = { which = gts_love_logs value < 0 } }
            set_variable = { which = gts_love_logs value = 0 }
        }
    }
}

#常规好感度系统关闭
gts_love_normal_system_end = {
    optimize_memory
    if = {
        limit = { has_leader_flag = has_gts_love_trait_normal }
        set_variable = { which = gts_love_logs value = 0 }
        remove_leader_flag = has_gts_love_trait
        remove_leader_flag = has_gts_love_trait_normal
        remove_trait = trait_leader_gts_love_logs_normal
    }
}

#好感度设置为特定值
gts_love_value_set = {
    if = {
        limit = { has_leader_flag = has_gts_love_trait }
        [[base_value]set_variable = { which =  gts_love_logs value = $base_value$ }]
        [[increase_value]set_variable = { which =  gts_normal_love_increase value = $increase_value$}]
    }
}
