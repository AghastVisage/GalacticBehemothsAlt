	possible = {

			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_no_existing_megastructure"
			has_no_non_gate_megastructure = yes
			NOT = {
				any_system_planet = {
					is_planet_class = pc_cosmogenesis_world
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
	}
	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_no_existing_gts_qinwen_megastructure_planets"
			from = {
				not = {
					count_owned_planet = {
						limit = {
							and = {
								has_planet_flag = pc_gts_qinwen_milk_planet
								is_planet_class = pc_gts_qinwen_milk_planet
							}
						}
						count >= 1
					}
				}
			}
		}
	}



	# root = system
	# from = country
	ai_weight = {
		factor = 0
	}

	on_build_start = {}
	on_build_cancel = {}
	on_build_complete = {
		owner = {
			country_event = {
				id = gts_qinwen.121
			}
		}
		from = { set_country_flag = godness_power_world_built }
		spawn_planet = {
			class = "pc_gts_qinwen_milk_planet"
			flags = {
				pc_gts_qinwen_milk_planet
			}
			location = fromfromfrom
			size = 5
			has_ring = no
			init_effect = {
				set_name = {
					key = pc_gts_qinwen_milk_planet
				}
				set_planet_entity = {
					entity = "gts_juice_planet_01_entity"
				}
				set_surveyed = {
					surveyed = yes
					surveyor = FROM
				}
				clear_blockers = yes

				set_planet_flag = megastructure
				set_planet_flag = godness_power_world_resettle@from

				fromfrom.planet = {
					set_planet_flag = has_megastructure
					set_planet_flag = has_godness_power_world
					if = {
						limit = { has_orbital_station = yes }
						orbital_station = {
							dismantle = yes
						}
					}
				}

				set_owner = fromfrom
				set_controller = fromfrom
				create_pop_group = {
					species = owner_main_species
				}
				from = {
					room_name_override = ""
				}
			}
		}
		remove_megastructure = fromfrom
	}
}
