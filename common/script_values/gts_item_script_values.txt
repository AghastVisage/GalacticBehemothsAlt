#饰品产出修正
GTS_LEADER_ITEM_MULT = {
	base = 1
	modifier = {
		add = 0.05
		OR = {
			AND = {
				is_scope_type = country
				has_technology = tech_item_phycics_1
			}
			AND = {
				NOT = {
					is_scope_type = country
				}
				exists = owner
				owner = {
					has_technology = tech_item_phycics_1
				}
			}
		}
	}
	modifier = {
		add = 0.05
		OR = {
			AND = {
				is_scope_type = country
				has_technology = tech_item_phycics_2
			}
			AND = {
				NOT = {
					is_scope_type = country
				}
				exists = owner
				owner = {
					has_technology = tech_item_phycics_2
				}
			}
		}
	}
	modifier = {
		add = 0.05
		OR = {
			AND = {
				is_scope_type = country
				has_technology = tech_item_phycics_3
			}
			AND = {
				NOT = {
					is_scope_type = country
				}
				exists = owner
				owner = {
					has_technology = tech_item_phycics_3
				}
			}
		}
	}
}
