gts_intel_back_effect = {
	# 向后翻页
	potential = {
		always = yes
	}
	allow = {
		hidden_trigger = {
			from = {
				check_variable = {
					which = gts_ui_intel_page
					value > 1
				}
			}
		}
	}
	effect = {
		custom_tooltip = gts_ui_back.tooltip
		from = {
			hidden_effect = {
				subtract_variable = {
					which = gts_ui_intel_page
					value = 1
				}
			}
		}
	}
}
