namespace = gts_enclave_event

# Separate dialogue and service hooks for each size tier.

# below_titan: dialogue menu.
country_event = {
    id = gts_enclave_event.310
    title = gts_enclave_event.310.title
    desc = gts_enclave_event.310.desc
    is_triggered_only = yes
    diplomatic = yes
    trigger = {
        exists = event_target:gts_enclave_contacting_country
        gts_enclave_is_below_titan = yes
    }
    picture_event_data = {
        portrait = event_target:gts_enclave_contacting_country
        room = enclave_curator_room
    }
    option = {
        name = gts_enclave_event.310.intro
        is_dialog_only = yes
        response_text = gts_enclave_event.310.intro.reply
    }
    option = {
        name = gts_enclave_event.310.services
        trigger = { gts_enclave_is_below_titan = yes }
        country_event = { id = gts_enclave_event.311 }
    }
    option = { name = gts_enclave_event.exit }
}

# below_titan: service interface. Future effects/modifiers belong in this tier only.
country_event = {
    id = gts_enclave_event.311
    title = gts_enclave_event.311.title
    desc = gts_enclave_event.311.desc
    is_triggered_only = yes
    diplomatic = yes
    trigger = {
        exists = event_target:gts_enclave_contacting_country
        gts_enclave_is_below_titan = yes
    }
    picture_event_data = {
        portrait = event_target:gts_enclave_contacting_country
        room = enclave_curator_room
    }
    option = {
        name = gts_enclave_event.311.order
        trigger = { gts_enclave_is_below_titan = yes }
        response_text = gts_enclave_event.311.order.reply
        allow = {
            resource_stockpile_compare = { resource = energy value >= 500 }
            custom_tooltip = {
                fail_text = gts_enclave_service.order.inactive
                NOT = { has_modifier = gts_enclave_below_titan_order }
            }
        }
        custom_tooltip = gts_enclave_event.311.order.tooltip
        hidden_effect = {
            add_resource = { energy = -500 }
            add_modifier = { modifier = gts_enclave_below_titan_order days = 3600 }
            event_target:gts_enclave_contacting_country = {
                add_trust = { who = root amount = 10 }
            }
            country_event = { id = gts_enclave_event.311 }
        }
    }
    option = {
        name = gts_enclave_event.311.production
        trigger = { gts_enclave_is_below_titan = yes }
        response_text = gts_enclave_event.311.production.reply
        allow = {
            resource_stockpile_compare = { resource = energy value >= 1500 }
            custom_tooltip = {
                fail_text = gts_enclave_service.production.inactive
                NOT = { has_modifier = gts_enclave_below_titan_production }
            }
            custom_tooltip = {
                fail_text = gts_enclave_service.requires_50_opinion
                event_target:gts_enclave_contacting_country = {
                    trust = { who = root value >= 50 }
                }
            }
        }
        custom_tooltip = gts_enclave_event.311.production.tooltip
        hidden_effect = {
            add_resource = { energy = -1500 }
            add_modifier = { modifier = gts_enclave_below_titan_production days = 3600 }
            country_event = { id = gts_enclave_event.311 }
        }
    }
    option = {
        name = gts_enclave_event.back
        country_event = { id = gts_enclave_event.300 }
    }
    option = { name = gts_enclave_event.exit }
}
