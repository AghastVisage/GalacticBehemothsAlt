# 開局
country_event = {

    id = gts_ultra_company.0
    hide_window = yes
    is_triggered_only = yes
    trigger = { has_origin = origin_Ultra_company }

    immediate = {

        # 避免初期領袖太多
        every_owned_leader = {
            limit = {
                is_ruler = no
            }
            kill_leader = {
                show_notification = no
            }
        }

        # 設置初期權利
        owner_main_species = {
            set_colonization_controls = {
                type = colonization_control_yes
            }
        }

        # 設置初期政策
        set_policy = {
            policy = gts_bruh_insurance_rate
            option = gts_bruh_insurance_rate_4
        }

        capital_scope = {

            # 設置星球特徵
            set_planet_size = 30
            clear_blockers = yes

            # 設置初期人口
            create_pop_group = {
                species = owner_main_species
                category = criminal
                size = 100
                effect = {
                    set_pop_group_flag = gts_bruh_ultra_company_begin_pop_group_flag
                }
            }
            every_owned_pop_group = {
                limit = {
                    NOT = {
                        has_pop_group_flag = gts_bruh_ultra_company_begin_pop_group_flag
                    }
                }
                kill_all_pop = yes
            }
            create_pop_group = {
                species = owner_main_species
                category = civilian
                size = 0
                effect = {
                    save_event_target_as = gts_bruh_ultra_company_civilian_pop_group
                }
            }
            every_owned_pop_group = {
                limit = {
                    has_pop_group_flag = gts_bruh_ultra_company_begin_pop_group_flag
                }
                remove_pop_group_flag = gts_bruh_ultra_company_begin_pop_group_flag
                transfer_pop_amount = {
                    source = This
                    target = event_target:gts_bruh_ultra_company_civilian_pop_group
                    percentage = 1
                }
            }
        }
    }

    after = {
        capital_scope = {

            # 設置初期陸軍
            create_fleet = {
                effect = {
                    set_owner = root
                    create_army_transport = {
                        ship_name = "NAME_Grand_Herald"
                        army_name = "NAME_Demon_Refugees"
                        army_type = "gts_big_army"
                        species = root.owner_main_species
                    }
                    set_location = {
                        target = prev
                    }
                }
            }

            # 設置初期建築與區劃
            remove_all_buildings = yes
            remove_all_districts = yes
            #remove_zone = {
            #    district = district_city
            #    zone = zone_research_unity
            #}
            #add_zone = {
            #    district = district_city
            #    zone = zone_unity
            #}
            add_building = building_colony_shelter
            add_building = building_shrink_pop_factory
            add_building = building_shrink_city_factory
            add_building = building_shrink_planet_factory
            add_building = building_transmutation_workshop
            add_building = building_commercial_zone
            add_building = {
                zone = zone_research_unity
                building = building_gts_bruh_administrative_tower
            }
            add_building = {
                zone = zone_research_unity
                building = building_research_lab_1
            }
            add_building = {
                zone = zone_industrial
                building = building_foundry_1
            }
            while = {
                count = 2
                add_deposit = d_frozen_gas_lake
                add_deposit = d_geothermal_vent
                add_deposit = d_prosperous_mesa
                add_deposit = d_rich_mountain
                add_deposit = d_lush_jungle
                add_deposit = d_black_soil
            }
            while = {
                count = 2
                add_building = {
                    zone = zone_industrial
                    building = building_factory_1
                }
            }
            while = {
                count = 5
                add_district = district_city
            }
            while = {
                count = 7
                add_district = district_generator
                add_district = district_farming
            }
            while = {
                count = 10
                add_district = district_mining
            }
        }
    }
}
