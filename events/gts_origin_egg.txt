country_event = {

	id = gts_egg.0
	hide_window = yes
	is_triggered_only = yes
	trigger = { has_origin = origin_Luanzi }

	immediate = {

		set_country_flag = Luanzi_country
		set_country_flag = starting_event

		capital_scope = {
			set_planet_flag = gts_hometown_planet
			gts_bruh_clear_planet_effect = yes
			remove_zone = {
				district = district_body_1
				zone = zone_urban
				zone_slot = 2
			}
			remove_building = building_precinct_house
			add_building = building_capital
			add_building = building_planet_jiansheshu_1
			add_building = building_active_weihuxitong_1
			add_building = building_holo_theatres
			add_building = building_factory_1
			#add_building = building_gts_bruh_administrative_tower
			if = {
				limit = {
					Root = {
						gts_is_tiny_empire = yes
					}
				}
				disable_building = building_planet_jiansheshu_1
				disable_building = building_active_weihuxitong_1
			}
			while = {
				count = 1
				add_district = district_body_5
				add_district = district_body_7
			}
			while = {
				count = 3
				add_district = district_body_1
			}
			while = {
				count = 5
				add_district = district_body_2
				add_district = district_body_4
			}
			while = {
				count = 7
				add_district = district_body_3
			}

			luanzi_origin_effect = yes
			save_global_event_target_as = gts_luanzi_planet
		}

		owner_main_species = {
			change_species_characteristics = {
				remove_trait = trait_pc_savannah_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_arid_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_desert_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_continental_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_tropical_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_ocean_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_arctic_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_tundra_preference
			}
			change_species_characteristics = {
				remove_trait = trait_pc_alpine_preference
			}
			change_species_characteristics = {
				add_trait = trait_pc_gts_luanzi_preference
			}
		}
	}
}
