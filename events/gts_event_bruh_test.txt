	id = gts_event_bruh_test.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_country = {
			is_ai = no
		}
	}

	immediate = {
		if = {
			limit = {
				any_country = {
					is_ai = no
					has_country_flag = gts_bruh_test_1_flag
				}
			}
			remove_country_flag = gts_bruh_test_1_flag
		}
		else = {
			random_country = {
				limit = {
					is_ai = no
				}
				set_country_flag = gts_bruh_test_1_flag
			}
		}
	}
}
