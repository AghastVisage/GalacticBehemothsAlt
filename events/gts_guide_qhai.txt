    }

        create_country = {
            name = NAME_qhai_communicate
            type = gts_qhai_commnunicate
            flag = {
                icon={
                    category="gts_country_flag"
                    file="flag_gts_qhai_B1.dds"
                }
                background={
                    category="backgrounds"
                    file="pattern_01.dds"
                }
                colors={
                    "dark_blue"
                    "black"

                }

        }
        last_created_country = {
            set_country_flag = protected_from_queen_storm
            establish_communications_no_message = root
            save_global_event_target_as = gts_qhai_country_target
        }
    }
}
