		}

			}
		}

		}
	}
	option = {
		name = gts_oversystem.1.d

		}

event = {
	id = gts_oversystem.3
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		any_system = { has_star_flag = oversystem_leather_shoes_system }
	}

	immediate = {
		random_system = {
			limit = { has_star_flag = oversystem_leather_shoes_system } # Shares spawn system with Tiyanki
			spawn_system = {
				min_distance = 3
				max_distance = 3
				max_jumps = 0
				initializer = "gts_oversystem_leather_shoes_system_2"
				hyperlane = no
				is_discovered = yes
				effect = {
				}
			}
			save_global_event_target_as = oversystem_leather_shoes_system
			random_neighbor_system = {
				set_star_flag = oversystem_leather_shoes_system_near
				random_system_planet = {
					add_anomaly = {
						category = gts_oversystem_leather_shoes_anomaly
					}
				}
			}
			every_neighbor_system = {
				limit = {
				}
				remove_hyperlane = {
					from = THIS
					to = prev
				}
			}
			random_system_planet = {
				create_country = {
					name = "NAME_oversystem_leather_shoes_system_country"
					type = oversystem_leather_shoes_communicate
					flag = {
						icon = {
							category = "zoological"
							file = "flag_zoological_1.dds"
						}
						background= {
							category = "backgrounds"
							file = "00_solid.dds"
						}
						colors={
							"black"
							"black"
							"null"
							"null"
						}
					}
					effect = {
						save_event_target_as = oversystem_leather_shoes_country

					}
				}
				while = {
					count = 4
					create_fleet = {
						name = "NAME_oversystem_leather_shoes_system_country"
						effect = {
							set_owner = event_target:oversystem_leather_shoes_country
							create_ship = {
								name = "NAME_oversystem_leather_shoes_system_country"
								design = "NAME_Cloud_Entity"
							}
							set_location = prev
							set_fleet_stance = aggressive
							set_aggro_range_measure_from = return_point
							set_aggro_range = 0
							queue_actions = {
								wait = { # Cooldown until they start moving
									duration = 2000
									random = 500
								}
							}
						}
					}
				}
			}
		}
	}
	after = {
	}
}
