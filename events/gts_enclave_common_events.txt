namespace = gts_enclave_common

# ROOT: discovering country. FROM: enclave.
country_event = {
    id = gts_enclave_common.1000
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        from = {
            is_country_type = gts_enclave
            has_country_flag = gts_enclave_country
        }
    }
    immediate = { establish_communications_no_message = from }
}

# Also handles communications received from another empire.
country_event = {
    id = gts_enclave_common.2000
    title = gts_enclave_common.2000.title
    desc = {
        trigger = { gts_enclave_is_below_titan = yes }
        text = gts_enclave_common.2000.desc.below_titan
    }
    desc = {
        trigger = { gts_enclave_is_titan = yes }
        text = gts_enclave_common.2000.desc.titan
    }
    desc = {
        trigger = { gts_enclave_is_ultra = yes }
        text = gts_enclave_common.2000.desc.ultra
    }
    picture = GFX_evt_communication_event
    is_triggered_only = yes
    trigger = {
        is_ai = no
        is_country_type = default
        NOT = { has_country_flag = has_comm_with_gts_enclave }
        from = {
            is_country_type = gts_enclave
            has_country_flag = gts_enclave_country
        }
    }
    immediate = {
        set_country_flag = has_comm_with_gts_enclave
        from = { save_event_target_as = gts_enclave_contacting_country }
    }
    option = {
        name = gts_enclave_common.2000.a
        country_event = { id = gts_enclave_event.21 }
    }
    option = { name = gts_enclave_common.2000.b }
}
