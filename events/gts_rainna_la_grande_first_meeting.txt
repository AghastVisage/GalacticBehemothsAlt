country_event = {
    id = gts_rainna_la_grande_first_meeting.00
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        if = {
            limit = {
                NOR = {
                    is_ai = yes
                    has_country_flag = rainna_la_grande_first_meeting_done
                }
            }
            if = {
                limit = {
                    count_planet_within_border = {
                        limit = {
                            has_owner = yes
                            is_colony = yes
                            rainna_la_grande_is_normal_planet = yes
                        }
                        count >= 2
                    }
                    any_planet_within_border = {
                        has_owner = yes
                        is_colony = yes
                        rainna_la_grande_is_normal_planet = yes
                    }
                }
                save_event_target_as = rainna_la_grande_first_meeting_country
                country_event = {
                    id = gts_rainna_la_grande_first_meeting.01
                }
            }
            else_if = {
                limit = {
                    is_nomadic = yes
                }
            }
            else = {
                country_event = {
                    id = gts_rainna_la_grande_first_meeting.00
                    years = 1
                    random = 180
                }
            }
        }
    }
}
