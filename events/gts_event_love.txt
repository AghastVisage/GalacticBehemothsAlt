namespace = gts_event_love

#五年一次全国好感度校验，增长
event = {
    id = gts_event_love.0
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        every_playable_country = {
            limit = {
                OR = {
                    is_ai = no
                    any_owned_leader = { has_leader_flag = has_gts_love_trait_normal }
                }
            }
            every_owned_leader = {
                limit = { has_leader_flag = has_gts_love_trait_normal }
                leader_event = {
                   id = gts_event_love.1
                }
                leader_event = {
                    id = gts_event_love.3
                }
            }
        }
    }
}
# 常规好感度增加
leader_event = {
    id = gts_event_love.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        NOT = {
            has_leader_flag = gts_love_auto_change_no
        }
        is_gts_love_leader = yes
        NAND = {
            NOT = { has_leader_flag = gts_married }
            check_variable = { which = gts_love_logs value >= gts_love_level5 }
        }
    }

    immediate = {
        gts_love_system_increase = yes
    }
}
